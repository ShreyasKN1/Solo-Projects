    int solve(int[] stoneValue, int i, int[] t)
    {
        int result=0;
        int n=stoneValue.length;
        if(i>=n) return 0;
        if(t[i]!=-1) return t[i];
        result=stoneValue[i]-solve(stoneValue,i+1,t);// alice takes 1 then remaining bob then alice
        if(i+1<n)
         {
            result=Math.max(result, stoneValue[i]+stoneValue[i+1]-solve(stoneValue,i+2,t)); // alice takes 2 then bob then so on build tree
         }
        if(i+2<n)
        {
         result=Math.max(result,stoneValue[i]+stoneValue[i+1]+stoneValue[i+2]-solve(stoneValue,i+3,t)); //alice can take 3 then bob remaining and build
        }
        t[i]=result;
        return t[i];
    }
}

    }

    boolean solve(int n, boolean[] t)
    {
        if(n==0)
        {
             return false; // whoever turn it was they lost
        }
        if(t[n]!=false) return t[n];
        for(int k=1;k*k<=n;k++)
        {
            if(solve(n-k*k,t)==false) //bob lost
            {
                return t[n]=true;//alice won
            }
        }  
          //no possible for alice to win  
        return t[n]=false;
    }
}
