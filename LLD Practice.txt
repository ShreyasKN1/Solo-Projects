Directory class:
class Directory extends FileDirectoryNode{

Map<String,FileDirectoryNode> hm=new HashMap<>();

public void add(FileDirectoryNode node)
{
		hm.put(node.name,node);
}
public FileDirectoryNode get(FileDirectoryNode node)
{
		hm.get(node.name);
}

@Override
public int getSize()
{
int size=0;
for(FileDirectory node:hm.values())
{
	size+=node.getSize();

}


}
